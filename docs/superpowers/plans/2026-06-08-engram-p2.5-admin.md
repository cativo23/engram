# Engram P2.5 — GitHub-OAuth Admin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
> **Plan lives OFF-REPO** (`~/engram-planning/plans/`) — never commit it. Repo: https://github.com/cativo23/engram

**Goal:** Add a GitHub-OAuth-gated `/admin` panel that lets the (allowlisted) owner log in with GitHub, list their own repos, and add/remove them from Engram's index at runtime — replacing the hardcoded `REPOS` list with a data-driven index.

**Architecture:** One GitHub OAuth flow serves as BOTH login and repo access. `/admin/login` → GitHub authorize (scope `read:user repo`, CSRF `state` in session) → callback validates state, exchanges code, fetches `/user`, checks the username against `ENGRAM_ADMIN_GITHUB_USERS` allowlist. On success: the GitHub access token is **Fernet-encrypted and stored in Postgres**, and a signed httpOnly session cookie (Starlette `SessionMiddleware`) authorizes `/admin/*`. Ingestion runs via FastAPI `BackgroundTasks` with per-repo status tracked on the `repos` row. A public read-only `GET /repos` feeds the INDEX CORE.

**Tech Stack:** FastAPI + Starlette `SessionMiddleware` (+ `itsdangerous`), raw `httpx` for the OAuth flow (no Authlib — fewer deps, explicit state control), `cryptography` Fernet for token-at-rest, psycopg3 + pgvector (existing), git-clone-with-token via a `credential.helper` that reads the token from env (never argv/`.git/config`).

```python
# engram/admin/crypto.py
"""Encrypt the GitHub access token at rest. The token is a `repo`-scoped
credential; if Postgres leaks, the attacker still needs FERNET_KEY (env, outside
the DB). Fernet = AES-128-CBC + HMAC-SHA256 (authenticated)."""

from cryptography.fernet import Fernet


def generate_key() -> str:
    """A new URL-safe base64 32-byte key. Run once; store in FERNET_KEY env."""
    return Fernet.generate_key().decode()


def encrypt_token(token: str, key: str) -> str:
    return Fernet(key.encode()).encrypt(token.encode()).decode()


def decrypt_token(ciphertext: str, key: str) -> str:
    return Fernet(key.encode()).decrypt(ciphertext.encode()).decode()
```

```python
from psycopg.rows import dict_row


def list_repos(conn) -> list[dict]:
    with conn.cursor(row_factory=dict_row) as cur:
        cur.execute(
            "SELECT id, slug, status, error, chunks, indexed_sha, indexed_at "
            "FROM repos ORDER BY slug"
        )
        return cur.fetchall()


def get_repo(conn, repo_id: int) -> dict | None:
    with conn.cursor(row_factory=dict_row) as cur:
        cur.execute(
            "SELECT id, slug, status, error, chunks, indexed_sha, indexed_at "
            "FROM repos WHERE id = %s", (repo_id,))
        return cur.fetchone()


def upsert_repo_queued(conn, slug: str) -> int:
    with conn.cursor() as cur:
        cur.execute(
            "INSERT INTO repos (slug, status) VALUES (%s, 'queued') "
            "ON CONFLICT (slug) DO UPDATE SET status='queued', error=NULL "
            "RETURNING id", (slug,))
        rid = cur.fetchone()[0]
    conn.commit()
    return rid


def set_repo_status(conn, repo_id: int, status: str, *,
                    error: str | None = None, indexed_sha: str | None = None,
                    chunks: int | None = None) -> None:
    with conn.cursor() as cur:
        cur.execute(
            "UPDATE repos SET status=%s, error=%s, "
            "indexed_sha=COALESCE(%s, indexed_sha), "
            "chunks=COALESCE(%s, chunks), "
            "indexed_at=CASE WHEN %s='done' THEN now() ELSE indexed_at END "
            "WHERE id=%s",
            (status, error, indexed_sha, chunks, status, repo_id))
    conn.commit()


def delete_repo(conn, repo_id: int) -> None:
    with conn.cursor() as cur:
        cur.execute("DELETE FROM repos WHERE id = %s", (repo_id,))  # chunks cascade
    conn.commit()


def save_github_credential(conn, login: str, encrypted_token: str) -> None:
    with conn.cursor() as cur:
        cur.execute(
            "INSERT INTO github_credentials (github_login, encrypted_token) "
            "VALUES (%s, %s) ON CONFLICT (github_login) DO UPDATE "
            "SET encrypted_token=EXCLUDED.encrypted_token, updated_at=now()",
            (login, encrypted_token))
    conn.commit()


def get_github_credential(conn, login: str) -> str | None:
    with conn.cursor() as cur:
        cur.execute(
            "SELECT encrypted_token FROM github_credentials WHERE github_login=%s",
            (login,))
        row = cur.fetchone()
        return row[0] if row else None
```

```python
# engram/admin/oauth.py
"""GitHub OAuth web flow + repo listing. Pure functions over an injectable
httpx.Client (tests pass a MockTransport; prod uses a default client). Never
log tokens here."""

from urllib.parse import urlencode

import httpx

AUTHORIZE_URL = "https://github.com/login/oauth/authorize"
TOKEN_URL = "https://github.com/login/oauth/access_token"
API_BASE = "https://api.github.com"
SCOPE = "read:user repo"
_API_HEADERS = {"Accept": "application/vnd.github+json",
                "X-GitHub-Api-Version": "2022-11-28"}


def build_authorize_url(*, client_id: str, redirect_uri: str, state: str) -> str:
    q = urlencode({
        "client_id": client_id, "redirect_uri": redirect_uri,
        "scope": SCOPE, "state": state,
    })
    return f"{AUTHORIZE_URL}?{q}"


def exchange_code(*, code: str, client_id: str, client_secret: str,
                  redirect_uri: str, client: httpx.Client | None = None) -> str:
    own = client is None
    client = client or httpx.Client(timeout=10)
    try:
        resp = client.post(TOKEN_URL, data={
            "client_id": client_id, "client_secret": client_secret,
            "code": code, "redirect_uri": redirect_uri,
        }, headers={"Accept": "application/json"})
        resp.raise_for_status()
        data = resp.json()
        if "access_token" not in data:
            raise ValueError(f"token exchange failed: {data.get('error', 'unknown')}")
        return data["access_token"]
    finally:
        if own:
            client.close()


def fetch_user_login(token: str, *, client: httpx.Client | None = None) -> str:
    own = client is None
    client = client or httpx.Client(timeout=10)
    try:
        resp = client.get(f"{API_BASE}/user",
                          headers={**_API_HEADERS, "Authorization": f"Bearer {token}"})
        resp.raise_for_status()
        return resp.json()["login"]
    finally:
        if own:
            client.close()


def list_owned_repos(token: str, *, client: httpx.Client | None = None) -> list[dict]:
    own = client is None
    client = client or httpx.Client(timeout=15)
    headers = {**_API_HEADERS, "Authorization": f"Bearer {token}"}
    url = f"{API_BASE}/user/repos?type=owner&visibility=all&per_page=100&page=1"
    out: list[dict] = []
    try:
        while url:
            resp = client.get(url, headers=headers)
            resp.raise_for_status()
            for r in resp.json():
                out.append({k: r.get(k) for k in
                            ("full_name", "private", "default_branch",
                             "description", "pushed_at")})
            url = _next_link(resp.headers.get("Link", ""))
        return out
    finally:
        if own:
            client.close()


def _next_link(link_header: str) -> str | None:
    for part in link_header.split(","):
        if 'rel="next"' in part:
            return part.split(";")[0].strip().strip("<>")
    return None
```

```python
session cookie + a GitHub-username allowlist."""

from fastapi import APIRouter, Depends, HTTPException, Request
from fastapi.responses import RedirectResponse

from ..config import settings
from ..db import connect, save_github_credential, get_github_credential
from . import oauth
from .crypto import encrypt_token, decrypt_token
from .security import is_allowlisted, require_admin

router = APIRouter(prefix="/admin")


# --- seams (monkeypatched in tests) ------------------------------------------
def _save_credential(login: str, token: str) -> None:
    enc = encrypt_token(token, settings.fernet_key)
    with connect() as conn:
        save_github_credential(conn, login, enc)


def _pop_state(request: Request, received: str) -> bool:
    expected = request.session.pop("oauth_state", None)
    return bool(expected) and secrets.compare_digest(expected, received)


# --- auth routes -------------------------------------------------------------
@router.get("/login")
def login(request: Request):
    state = secrets.token_urlsafe(32)
    request.session["oauth_state"] = state
    url = oauth.build_authorize_url(
        client_id=settings.github_client_id,
        redirect_uri=settings.oauth_redirect_uri,
        state=state,
    )
    return RedirectResponse(url)


@router.get("/auth/callback")
def auth_callback(request: Request, code: str = "", state: str = ""):
    if not _pop_state(request, state):
        raise HTTPException(status_code=400, detail="state mismatch")
    token = oauth.exchange_code(
        code=code, client_id=settings.github_client_id,
        client_secret=settings.github_client_secret,
        redirect_uri=settings.oauth_redirect_uri,
    )
    login_name = oauth.fetch_user_login(token)
    if not is_allowlisted(login_name, settings.admin_github_users):
        raise HTTPException(status_code=403, detail="not authorized")
    _save_credential(login_name, token)
    request.session["user"] = login_name
    return RedirectResponse(url="/admin", status_code=302)


@router.post("/logout")
def logout(request: Request, user: str = Depends(require_admin)):
    request.session.clear()
```

```python
def _token_for(user: str) -> str:
    with connect() as conn:
        enc = get_github_credential(conn, user)
    if not enc:
        raise HTTPException(status_code=400, detail="no GitHub token on file; re-login")

def _run_ingestion(repo_id: int, slug: str, user: str) -> None:
    """Background task: clone+embed with the user's token, tracking status."""
    try:

        token = _token_for(user)

    except Exception as exc:                       # noqa: BLE001 — record, don't crash worker
        with connect() as conn:
            set_repo_status(conn, repo_id, "failed", error=str(exc)[:500])
```

```javascript
// engram/web/admin-state.js
// Pure helpers for the admin page. No DOM, no fetch — unit-tested.

export function mergeGithubAndIndexed(githubRepos, indexedRepos) {
  const bySlug = new Map(indexedRepos.map((r) => [r.slug, r]));
  return githubRepos.map((g) => {
    const idx = bySlug.get(g.full_name);
    return {
      full_name: g.full_name,
      private: !!g.private,
      indexed: !!idx,
      id: idx ? idx.id : null,
      status: idx ? idx.status : null,
      chunks: idx ? idx.chunks : 0,
    };
  });
}

export function statusBadge(status) {
  switch (status) {
    case "done": return "INDEXED";
    case "running": return "INDEXING…";
    case "queued": return "QUEUED";
    case "failed": return "FAILED";
    default: return "— NOT INDEXED";
  }
}

export function isBusy(repos) {
  return repos.some((r) => r.status === "queued" || r.status === "running");
}
```

```javascript
// engram/web/admin.js
import { mergeGithubAndIndexed, statusBadge, isBusy } from "./admin-state.js";

const $ = (id) => document.getElementById(id);

async function api(path, opts = {}) {
  const resp = await fetch(path, { credentials: "same-origin", ...opts });
  return resp;
}

async function loadState() {
  // /admin/repos returns 401 if not logged in → show login zone.
  const indexedResp = await api("/admin/repos");
  if (indexedResp.status === 401) {
    $("login-zone").hidden = false;
    $("manage-zone").hidden = true;
    return;
  }
  $("login-zone").hidden = true;
  $("manage-zone").hidden = false;

  const indexed = await indexedResp.json();
  let github = [];
  try {
    const gh = await api("/admin/github/repos");
    if (gh.ok) github = await gh.json();
  } catch { /* token may be missing; still show indexed */ }

  $("who").textContent = github.length || indexed.length ? "AUTHENTICATED" : "";
  renderIndexed(indexed);
  renderGithub(mergeGithubAndIndexed(github, indexed));

  if (isBusy(indexed)) setTimeout(loadState, 2000); // poll while indexing
}

function renderIndexed(rows) {
  $("indexed-list").innerHTML = rows.map((r) => `
    <li>
      <span class="slug">${escapeHtml(r.slug)}</span>
      <span class="badge ${r.status}">${statusBadge(r.status)}</span>
      <span class="muted">${r.chunks} chunks</span>
      ${r.error ? `<span class="err">${escapeHtml(r.error)}</span>` : ""}
      <button data-del="${r.id}" class="btn ghost">REMOVE</button>
    </li>`).join("");
  $("indexed-list").querySelectorAll("[data-del]").forEach((b) =>
    b.addEventListener("click", () => removeRepo(b.dataset.del)));
}

function renderGithub(rows) {
  $("github-list").innerHTML = rows.map((r) => `
    <li>
      <span class="slug">${escapeHtml(r.full_name)}</span>
      ${r.private ? '<span class="badge private">PRIVATE</span>' : ""}
      ${r.indexed
        ? `<span class="badge ${r.status}">${statusBadge(r.status)}</span>`
        : `<button data-add="${escapeHtml(r.full_name)}" class="btn">INDEX</button>`}
    </li>`).join("");
  $("github-list").querySelectorAll("[data-add]").forEach((b) =>
    b.addEventListener("click", () => addRepo(b.dataset.add)));
}

async function addRepo(slug) {
  $("add-err").textContent = "";
  const resp = await api("/admin/repos", {
    method: "POST", headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ slug }),
  });
  if (!resp.ok) {
    $("add-err").textContent = `Failed to queue ${slug} (${resp.status})`;
    return;
  }
  loadState();
}

async function removeRepo(id) {
  await api(`/admin/repos/${id}`, { method: "DELETE" });
  loadState();
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, (c) => ({
    "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]));
}

$("add-form").addEventListener("submit", (e) => {
  e.preventDefault();
  const slug = $("slug-input").value.trim();
  if (slug) addRepo(slug);
});
$("logout").addEventListener("click", async () => {
  await api("/admin/logout", { method: "POST" });
  location.reload();
});

loadState();
```

```javascript
// engram/web/app-repos.js
export function reposToIndexCore(payload) {
  if (!Array.isArray(payload)) return [];
  return payload.map((r) => ({
    slug: r.slug,
    name: r.slug.split("/").pop(),
    chunks: r.chunks || 0,
  }));
}
```
